                const color = generateColor();
                box.style.backgroundColor = color;
                box.textContent = `color: ${color}`
            })
            const generateColor = () => {
                const letters = "0123456789ABCDEF"
                let color = "#"
                for (let i = 0; i < 6; i++) {
                    color += letters[Math.floor(Math.random() * 16)];
                }
                return color;
            }
